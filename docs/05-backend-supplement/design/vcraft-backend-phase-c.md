```java
package vn.vcraft.backend.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;
import org.springframework.web.filter.CorsFilter;

import java.util.List;

@Configuration
public class CorsConfig {

    @Value("${cors.allowed-origins:http://localhost:5173}")
    private String allowedOrigins;

    @Bean
    public CorsFilter corsFilter() {
        var config = new CorsConfiguration();
        config.setAllowedOrigins(List.of(allowedOrigins.split(",")));
        config.setAllowedMethods(List.of("GET","POST","PUT","PATCH","DELETE","OPTIONS"));
        config.setAllowedHeaders(List.of("Authorization","Content-Type","X-Requested-With"));
        config.setAllowCredentials(true);
        config.setMaxAge(3600L);

        var source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/api/**", config);

        return new CorsFilter(source);
    }
}
```

### 0.7. SlugUtils.java — File mới

**Package:** `vn.vcraft.backend.validation`

```java
package vn.vcraft.backend.validation;

import java.text.Normalizer;
import java.util.function.Predicate;

public final class SlugUtils {

    private SlugUtils() {}

    /**
     * Chuyển chuỗi tiếng Việt thành slug URL-safe.
     * Ví dụ: "Gốm Bát Tràng" → "gom-bat-trang"
     */
    public static String toSlug(String input) {
        if (input == null || input.isBlank()) return "";
        // 1. Normalize Unicode NFD để tách dấu ra
        String normalized = Normalizer.normalize(input.strip(), Normalizer.Form.NFD);
        // 2. Xóa combining marks (dấu thanh, dấu mũ...)
        String noMark = normalized.replaceAll("\\p{InCombiningDiacriticalMarks}+", "");
        // 3. Xử lý đặc biệt: đ → d, Đ → D (không bị NFD tách)
        String noDdash = noMark.replace("đ", "d").replace("Đ", "D");
        // 4. Lowercase, thay ký tự không phải a-z0-9 bằng dấu gạch ngang
        String slug = noDdash.toLowerCase()
                .replaceAll("[^a-z0-9]+", "-")
                .replaceAll("^-|-$", "");
        return slug.isEmpty() ? "item" : slug;
    }

    /**
     * Sinh slug duy nhất bằng cách thêm hậu tố nếu trùng.
     * Caller cung cấp Predicate kiểm tra xem slug đã tồn tại chưa.
     */
    public static String uniqueSlug(String base, Predicate<String> exists) {
        String candidate = toSlug(base);
        if (!exists.test(candidate)) return candidate;
        for (int i = 2; i <= 999; i++) {
            String with = candidate + "-" + i;
            if (!exists.test(with)) return with;
        }
        // Fallback: thêm timestamp (không xảy ra trong thực tế)
        return candidate + "-" + System.currentTimeMillis();
    }
}
```

### 1.3. OtpService.java — File mới

**Package:** `vn.vcraft.backend.service`

```java
package vn.vcraft.backend.service;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.mail.SimpleMailMessage;
import org.springframework.mail.javamail.JavaMailSender;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.vcraft.backend.exception.ApiException;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.SecureRandom;
import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.HexFormat;

@Service
public class OtpService {

    private final JavaMailSender mailer;
    private final OtpRepository otpRepo;  // JDBC repository — xem 1.4

    @Value("${auth.otp-minutes:15}")
    private int otpMinutes;

    public OtpService(JavaMailSender mailer, OtpRepository otpRepo) {
        this.mailer = mailer;
        this.otpRepo = otpRepo;
    }

    /** Sinh OTP 6 chữ số, lưu hash vào DB, gửi email. */
    @Transactional
    public void sendEmailOtp(long accountId, String toEmail, String purpose) {
        // Hủy OTP cũ chưa dùng của cùng account + purpose
        otpRepo.invalidateExisting(accountId, purpose);

        String otp = String.format("%06d", new SecureRandom().nextInt(1_000_000));
        String hash = sha256(otp);
        Instant expiresAt = Instant.now().plus(otpMinutes, ChronoUnit.MINUTES);

        otpRepo.create(accountId, hash, purpose, expiresAt);

        // Gửi email (async trong production — đơn giản hóa cho bài tập)
        var msg = new SimpleMailMessage();
        msg.setTo(toEmail);
        msg.setSubject("[vCraft] Mã xác thực email");
        msg.setText("Mã xác thực của bạn là: " + otp +
                    "\nMã có hiệu lực trong " + otpMinutes + " phút." +
                    "\nNếu bạn không yêu cầu, hãy bỏ qua email này.");
        mailer.send(msg);
    }

    /** Xác thực OTP; đánh dấu đã dùng. Ném ApiException nếu sai/hết hạn. */
    @Transactional
    public void verifyOtp(long accountId, String otp, String purpose) {
        String hash = sha256(otp);
        boolean valid = otpRepo.markUsed(accountId, hash, purpose, Instant.now());
        if (!valid) {
            throw new ApiException(
                org.springframework.http.HttpStatus.BAD_REQUEST,
                "INVALID_OTP",
                "Mã xác thực không đúng hoặc đã hết hạn.");
        }
    }

    private static String sha256(String input) {
        try {
            return HexFormat.of().formatHex(
                MessageDigest.getInstance("SHA-256")
                    .digest(input.getBytes(StandardCharsets.UTF_8)));
        } catch (Exception e) {
            throw new IllegalStateException(e);
        }
    }
}
```

### 1.4. OtpRepository.java — File mới

```java
package vn.vcraft.backend.repository;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;
import java.time.Instant;

@Repository
public class OtpRepository {
    private final JdbcTemplate db;

    public OtpRepository(JdbcTemplate db) { this.db = db; }

    public void invalidateExisting(long accountId, String purpose) {
        db.update("""
            UPDATE email_otps SET used_at = now()
            WHERE account_id = ? AND purpose = ? AND used_at IS NULL
            """, accountId, purpose);
    }

    public void create(long accountId, String otpHash, String purpose, Instant expiresAt) {
        db.update("""
            INSERT INTO email_otps(account_id, otp_hash, purpose, expires_at)
            VALUES (?, ?, ?, ?)
            """, accountId, otpHash, purpose, expiresAt);
    }

    /** Đánh dấu OTP đã dùng nếu hợp lệ. Trả true nếu thành công. */
    public boolean markUsed(long accountId, String otpHash, String purpose, Instant now) {
        int rows = db.update("""
            UPDATE email_otps
            SET used_at = now()
            WHERE account_id = ? AND otp_hash = ? AND purpose = ?
              AND expires_at > ? AND used_at IS NULL
            """, accountId, otpHash, purpose, now);
        return rows > 0;
    }
}
```

```java
    @Value("${app.base-url:http://localhost:8080}")
    private String baseUrl;

    @Override
    public String store(byte[] data, String filename) {
        try {
            LocalDate now = LocalDate.now();
            // Tổ chức theo năm/tháng để tránh thư mục quá nhiều file
            Path dir = Path.of(storagePath,
                String.valueOf(now.getYear()),
                String.format("%02d", now.getMonthValue()));
            Files.createDirectories(dir);

            String stored = UUID.randomUUID().toString().replace("-","") + "_" + filename;
            Files.write(dir.resolve(stored), data);

            return now.getYear() + "/" + String.format("%02d", now.getMonthValue())
                   + "/" + stored;
        } catch (IOException e) {
            throw new RuntimeException("Lỗi lưu ảnh: " + e.getMessage(), e);
        }
    }

    @Override
    public void delete(String relativePath) {
        try {
            Files.deleteIfExists(Path.of(storagePath, relativePath));
        } catch (IOException e) {
            // Log và tiếp tục — không block request
        }
    }

    @Override
    public String toUrl(String relativePath) {
        return baseUrl + "/media/" + relativePath;
    }
```

```java
package vn.vcraft.backend.service;

import net.coobird.thumbnailator.Thumbnails;
import org.apache.tika.Tika;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;
import vn.vcraft.backend.dto.MediaResponse;
import vn.vcraft.backend.exception.ApiException;
import vn.vcraft.backend.model.Actor;
import vn.vcraft.backend.repository.MediaRepository;
import vn.vcraft.backend.storage.MediaStorage;

import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Set;

@Service
public class MediaService {

    private static final Set<String> ALLOWED_TYPES =
        Set.of("image/jpeg", "image/png", "image/webp");
    private static final Tika TIKA = new Tika();

    @Value("${media.max-file-size-mb:5}")
    private int maxFileSizeMb;

    @Value("${media.max-files-per-request:10}")
    private int maxFilesPerRequest;

    private final MediaStorage storage;
    private final MediaRepository mediaRepo;

    public MediaService(MediaStorage storage, MediaRepository mediaRepo) {
        this.storage = storage;
        this.mediaRepo = mediaRepo;
    }

    @Transactional
    public List<MediaResponse> upload(Actor actor, List<MultipartFile> files) {
        if (files == null || files.isEmpty()) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "VALIDATION_FAILED",
                "Vui lòng chọn ít nhất một ảnh.");
        }
        if (files.size() > maxFilesPerRequest) {
            throw new ApiException(HttpStatus.BAD_REQUEST, "VALIDATION_FAILED",
                "Tối đa " + maxFilesPerRequest + " ảnh mỗi lần.");
        }

        List<MediaResponse> results = new ArrayList<>();
        for (MultipartFile file : files) {
            results.add(processOne(actor, file));
        }
        return results;
    }

    private MediaResponse processOne(Actor actor, MultipartFile file) {
        // 1. Kiểm tra kích thước
        long maxBytes = (long) maxFileSizeMb * 1024 * 1024;
        if (file.getSize() > maxBytes) {
            throw new ApiException(HttpStatus.PAYLOAD_TOO_LARGE, "UPLOAD_TOO_LARGE",
                "Ảnh phải nhỏ hơn " + maxFileSizeMb + " MB.");
        }

        // 2. Validate MIME bằng Tika (magic bytes)
        try {
            byte[] bytes = file.getBytes();
            String detectedType = TIKA.detect(bytes);
            if (!ALLOWED_TYPES.contains(detectedType)) {
                throw new ApiException(HttpStatus.UNSUPPORTED_MEDIA_TYPE,
                    "UNSUPPORTED_MEDIA_TYPE",
                    "Chỉ chấp nhận ảnh JPEG, PNG hoặc WebP.");
            }

            // 3. Đọc ảnh gốc
            BufferedImage original = ImageIO.read(file.getInputStream());
            if (original == null) {
                throw new ApiException(HttpStatus.BAD_REQUEST, "INVALID_INPUT",
                    "Không thể đọc file ảnh.");
            }
            int origW = original.getWidth(), origH = original.getHeight();

            // 4. Sinh 3 biến thể
            String ext = detectedType.equals("image/png") ? "png" : "jpg";
            String baseName = "img." + ext;
            String outputFormat = ext.equals("png") ? "png" : "jpg";

            String pathThumb  = storage.store(resize(original, 240,  outputFormat), "thumb_"  + baseName);
            String pathMedium = storage.store(resize(original, 800,  outputFormat), "medium_" + baseName);
            String pathLarge  = storage.store(resize(original, 1600, outputFormat), "large_"  + baseName);

            // 5. Lưu metadata vào DB (trong @Transactional của caller)
            long mediaId = mediaRepo.create(
                actor.id(),
                file.getOriginalFilename() != null ? file.getOriginalFilename() : "image",
                detectedType,
                (int) file.getSize(),
                origW, origH,
                pathThumb, pathMedium, pathLarge
            );

            return new MediaResponse(mediaId,
                storage.toUrl(pathThumb),
                storage.toUrl(pathMedium),
                storage.toUrl(pathLarge),
                origW, origH);

        } catch (IOException e) {
            throw new ApiException(HttpStatus.INTERNAL_SERVER_ERROR, "INTERNAL_ERROR",
                "Lỗi xử lý ảnh.");
        }
    }

    private byte[] resize(BufferedImage src, int maxWidth, String format) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Thumbnails.of(src)
            .width(Math.min(src.getWidth(), maxWidth))  // không phóng to ảnh nhỏ
            .outputFormat(format)
            .outputQuality(0.85)
            .toOutputStream(out);
        return out.toByteArray();
    }

    @Transactional
    public void delete(Actor actor, long mediaId) {
        var media = mediaRepo.findById(mediaId)
            .orElseThrow(() -> new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                "Không tìm thấy ảnh."));

        if (media.ownerAccountId() != actor.id()) {
            throw new ApiException(HttpStatus.FORBIDDEN, "FORBIDDEN",
                "Bạn không có quyền xóa ảnh này.");
        }

        // Kiểm tra ảnh chưa được gắn vào thực thể nào
        if (mediaRepo.isLinked(mediaId)) {
            throw new ApiException(HttpStatus.CONFLICT, "DATA_CONFLICT",
                "Ảnh đang được sử dụng, không thể xóa.");
        }

        mediaRepo.delete(mediaId);
        storage.delete(media.pathThumb());
        storage.delete(media.pathMedium());
        storage.delete(media.pathLarge());
    }
}
```

```java
            """, Long.class,

    }

    public Optional<MediaRow> findById(long id) {
        return db.query("""
            SELECT id, owner_account_id, path_thumb, path_medium, path_large
            FROM media WHERE id = ?
            """,

                rs.getString("path_thumb"), rs.getString("path_medium"), rs.getString("path_large")),
            id).stream().findFirst();

        var count = db.queryForObject(
            "SELECT COUNT(*) FROM media_links WHERE media_id = ?", Integer.class, mediaId);
        return count != null && count > 0;
    }

    public void delete(long mediaId) {
        db.update("DELETE FROM media WHERE id = ?", mediaId);
    }

    /** Thay thế toàn bộ media_links của một entity. */
    public void replaceLinks(String entityType, long entityId, java.util.List<Long> mediaIds) {
        db.update("DELETE FROM media_links WHERE entity_type = ? AND entity_id = ?",
            entityType, entityId);
        for (int i = 0; i < mediaIds.size(); i++) {
            db.update("""
                INSERT INTO media_links(media_id, entity_type, entity_id, position)
                VALUES (?, ?, ?, ?)
                ON CONFLICT (media_id, entity_type, entity_id) DO UPDATE SET position = EXCLUDED.position
                """, mediaIds.get(i), entityType, entityId, i);
        }
    }

    /** Lấy ảnh của một entity, sắp theo position. */
    public java.util.List<MediaResponse> findByEntity(String entityType, long entityId,
                                                        String baseUrl) {
        return db.query("""
            SELECT m.id, m.path_thumb, m.path_medium, m.path_large, m.width, m.height
            FROM media m JOIN media_links ml ON ml.media_id = m.id
            WHERE ml.entity_type = ? AND ml.entity_id = ?
            ORDER BY ml.position

                rs.getLong("id"),
                baseUrl + "/media/" + rs.getString("path_thumb"),
                baseUrl + "/media/" + rs.getString("path_medium"),
```

```java
package vn.vcraft.backend.service;

import com.google.zxing.*;
import com.google.zxing.client.j2se.MatrixToImageWriter;
import com.google.zxing.common.BitMatrix;
import com.google.zxing.qrcode.QRCodeWriter;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.vcraft.backend.exception.ApiException;
import vn.vcraft.backend.model.*;

import vn.vcraft.backend.dto.*;

import java.io.ByteArrayOutputStream;
import java.security.SecureRandom;
import java.time.Instant;
import java.util.List;

@Service
public class BatchService {

    private static final String PUBLIC_CODE_CHARS =
        "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    private static final int PUBLIC_CODE_LEN = 10;

    @Value("${app.base-url:http://localhost:8080}")
    private String baseUrl;

    private final BatchRepository batchRepo;
    private final MediaRepository mediaRepo;
    private final ProductRepository productRepo;

    public BatchService(BatchRepository batchRepo, MediaRepository mediaRepo,
                        ProductRepository productRepo) {
        this.batchRepo = batchRepo;
        this.mediaRepo = mediaRepo;
        this.productRepo = productRepo;
    }

    /** GET /api/batches/{publicCode}?token= */
    public BatchDetailResponse getLandingPage(String publicCode, String token) {
        var batch = batchRepo.findByPublicCode(publicCode)
            .orElseThrow(() -> new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                "Không tìm thấy lô hàng."));

        if ("PRIVATE".equals(batch.getVisibility())) {
            if (token == null || !token.equals(batch.getShareToken())) {
                // Trả 404 — không lộ sự tồn tại của lô riêng tư
                throw new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                    "Không tìm thấy lô hàng.");
            }
        }
        if (Boolean.TRUE.equals(batch.getHidden())) {
            throw new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                "Không tìm thấy lô hàng.");
        }

        return toDetail(batch);
    }

    @Transactional
    public BatchDetailResponse create(Actor actor, BatchCreateInput input) {
        long workshopId = requireWorkshop(actor);

        // Kiểm tra sản phẩm thuộc xưởng
        var product = productRepo.findById(input.productId())
            .orElseThrow(() -> new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                "Không tìm thấy sản phẩm."));
        if (!product.getWorkshop().getId().equals(workshopId)) {
            throw new ApiException(HttpStatus.FORBIDDEN, "FORBIDDEN",
                "Sản phẩm không thuộc xưởng của bạn.");
        }

        String publicCode = generateUniquePublicCode();

        Batch batch = new Batch();
        // Set fields từ input...
        batch.setPublicCode(publicCode);
        batch.setWorkshopId(workshopId);
        batch.setCreatedAt(Instant.now());
        batch.setUpdatedAt(Instant.now());

        batchRepo.save(batch);

        if (input.mediaIds() != null) {
            mediaRepo.replaceLinks("BATCH", batch.getId(), input.mediaIds());
        }
        return toDetail(batch);
    }

    /** POST /api/artisan/batches/{id}/share-token — tạo liên kết riêng */
    @Transactional
    public String generateShareToken(Actor actor, long batchId) {
        var batch = ownedBatch(actor, batchId);
        // Sinh token 32 char URL-safe
        String token = generateToken(32);
        batch.setShareToken(token);
        batch.setUpdatedAt(Instant.now());
        batchRepo.save(batch);
        // Trả URL đầy đủ để Frontend hiển thị
        return baseUrl + "/batches/" + batch.getPublicCode() + "?token=" + token;
    }

    /** DELETE /api/artisan/batches/{id}/share-token — thu hồi */
    @Transactional
    public void revokeShareToken(Actor actor, long batchId) {
        var batch = ownedBatch(actor, batchId);
        batch.setShareToken(null);
        batch.setUpdatedAt(Instant.now());
        batchRepo.save(batch);
    }

    /** GET /api/batches/{publicCode}/qr.png — QR lô công khai */
    public byte[] generatePublicQr(String publicCode) throws Exception {
        var batch = batchRepo.findByPublicCode(publicCode)
            .orElseThrow(() -> new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                "Không tìm thấy lô hàng."));
        if (!"PUBLIC".equals(batch.getVisibility())) {
            throw new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",
                "Không tìm thấy lô hàng.");
        }
        String url = baseUrl + "/batches/" + publicCode;
        return encodeQr(url);
    }

    /** GET /api/artisan/batches/{id}/qr.png — QR kể cả lô riêng tư */
    public byte[] generateArtisanQr(Actor actor, long batchId) throws Exception {
        var batch = ownedBatch(actor, batchId);
        String url;
        if ("PRIVATE".equals(batch.getVisibility()) && batch.getShareToken() != null) {
            url = baseUrl + "/batches/" + batch.getPublicCode()
                + "?token=" + batch.getShareToken();
        } else {
            url = baseUrl + "/batches/" + batch.getPublicCode();
        }
        return encodeQr(url);
    }

    /** PUT /api/artisan/batches/{id}/milestones */
    @Transactional
    public void updateMilestones(Actor actor, long batchId, List<MilestoneInput> milestones) {
        var batch = ownedBatch(actor, batchId);
        batchRepo.replaceMilestones(batch.getId(), milestones, mediaRepo);
    }

    // ---- Helper ----

    private byte[] encodeQr(String url) throws Exception {
        QRCodeWriter writer = new QRCodeWriter();
        BitMatrix matrix = writer.encode(url, BarcodeFormat.QR_CODE, 300, 300);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        MatrixToImageWriter.writeToStream(matrix, "PNG", out);
        return out.toByteArray();
    }

    private String generateUniquePublicCode() {
        for (int attempt = 0; attempt < 5; attempt++) {
            String code = randomCode(PUBLIC_CODE_LEN);
            if (!batchRepo.existsByPublicCode(code)) return code;
        }
        throw new IllegalStateException("Không thể sinh mã lô duy nhất.");
    }

    private String generateToken(int len) {
        return randomCode(len);
    }

    private static String randomCode(int len) {
        SecureRandom rnd = new SecureRandom();
        StringBuilder sb = new StringBuilder(len);
        for (int i = 0; i < len; i++) {
            sb.append(PUBLIC_CODE_CHARS.charAt(rnd.nextInt(PUBLIC_CODE_CHARS.length())));
        }
        return sb.toString();
    }

    private Batch ownedBatch(Actor actor, long batchId) {
        var batch = batchRepo.findById(batchId)
            .orElseThrow(() -> new ApiException(HttpStatus.NOT_FOUND, "NOT_FOUND",

            throw new ApiException(HttpStatus.FORBIDDEN, "FORBIDDEN",
                "Lô không thuộc xưởng của bạn.");
        }
        return batch;
    }

            throw new ApiException(HttpStatus.FORBIDDEN, "FORBIDDEN",
```

```java
    /** POST /api/admin/workshops/{id}/lock */

            .orElseThrow(() -> new ApiException(NOT_FOUND, "NOT_FOUND", "Không tìm thấy xưởng."));
```

```java
public boolean isEmailVerified(long accountId) {
    var count = db.queryForObject("""
        SELECT COUNT(*) FROM accounts
        WHERE id = ? AND email_verified_at IS NOT NULL
        """, Integer.class, accountId);
    return count != null && count > 0;
}
```
